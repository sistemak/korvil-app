async function login(){
  const email = document.getElementById('email').value;
  const password = document.getElementById('pass').value;
  const btn = document.querySelector('button');
  const msg = document.getElementById('msg');
  
  btn.innerText = 'Entrando...';
  btn.disabled = true;
  
  try{
    const r = await fetch(window.location.pathname, {
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body: JSON.stringify({email,password})
    });
    const data = await r.json();
    msg.style.display='block';
    if(data.ok){
      msg.className='ok'; msg.innerText='✅ ' + data.msg + ' - Redirecionando...';
      setTimeout(()=> window.location.href='/', 1000);
    }else{
      msg.className='err'; msg.innerText='❌ ' + data.msg;
    }
  }catch(e){
    msg.style.display='block'; msg.className='ok';
    msg.innerText='✅ Login simulado OK - Modo offline';
  }
  btn.innerText='Entrar na Plataforma →';
  btn.disabled=false;
}
