// korvil/sections/korvil-loja/index.html - Lógica integrada de criação + commit automático // COLE NO SEU App.tsx / index.html da loja const REPO = "sistemak/korvil-app"; const BRANCH = "main"; function removerAcentos(s){ return s.normalize('NFD').replace(/[\u0300-\u036f]/g,''); } function gerarNomePasta(nome){ const partes = removerAcentos(nome.trim()).toLowerCase().split(/\s+/).filter(Boolean); if(partes.length<=1) return partes[0]||''; return `${partes[0]}_${partes.slice(1).map(p=>p[0]).join('')}`.replace(/[^a-z0-9_]/g,'').slice(0,40); } function validarDadosCompleto({nome,email,senha,confirmar}){ if(!nome || !email || !senha || !confirmar) return {ok:false, erro:'Preencha todos os dados'}; if(nome.trim().length<3) return {ok:false, erro:'Nome >=3 caracteres'}; if(!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) return {ok:false, erro:'Email inválido'}; if(senha.length<6) return {ok:false, erro:'Senha >=6'}; if(senha!==confirmar) return {ok:false, erro:'Senhas não conferem'}; return {ok:true}; } async function criarContaReal({nome,email,senha,confirmar}){ const valid = validarDadosCompleto({nome,email,senha,confirmar}); if(!valid.ok){ alert(valid.erro); return {ok:false, erro: valid.erro}; } const token = localStorage.getItem('loja_github_token') || localStorage.getItem('gh_token'); if(!token){ alert('Configure GH_TOKEN no admin primeiro'); return {ok:false}; } const pastaMae = gerarNomePasta(nome); const path = `korvil/sections/korvil-loja/login/contas/${pastaMae}/${pastaMae}.json`; // 1. Verifica se já existe - nunca duplicar try{ const check = await fetch(`https://api.github.com/repos/${REPO}/contents/${path}?ref=${BRANCH}`,{ headers:{ Authorization: `Bearer ${token}`, Accept:'application/vnd.github.v3+json' } }); if(check.ok){ alert('Conta já existe: '+pastaMae); return {ok:false, erro:'já existe'}; } }catch(_){} // 2. Cria JSON com mesmo nome da pasta mãe const conta = { id: (crypto.randomUUID ? crypto.randomUUID() : Date.now()+'_'+Math.random()), pasta: pastaMae, nome: nome.trim(), email: email.trim().toLowerCase(), senha_hash: await sha256(senha), criado_em: new Date().toISOString() }; // 3. Commit automático via GitHub API - cria pasta mãe/.json const putRes = await fetch(`https://api.github.com/repos/${REPO}/contents/${path}`,{ method:'PUT', headers:{ Authorization: `Bearer ${token}`, Accept:'application/vnd.github.v3+json', 'Content-Type':'application/json' }, body: JSON.stringify({ message: `KORVIL LOJA: cria conta ${pastaMae}`, content: btoa(unescape(encodeURIComponent(JSON.stringify(conta,null,2)))), branch: BRANCH }) }); if(!putRes.ok){ const t=await putRes.text(); throw new Error('Falha commit: '+t); } alert('Conta criada: '+pastaMae); return {ok:true, pastaMae}; } async function sha256(str){ const buf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(str)); return Array.from(new Uint8Array(buf)).map(b=>b.toString(16).padStart(2,'0')).join(''); } // Uso no botão da tela criação: // document.getElementById('btnCriar').onclick = async ()=>{ // const dados = { // nome: document.getElementById('nome').value, // email: document.getElementById('email').value, // senha: document.getElementById('senha').value, // confirmar: document.getElementById('confirmar').value // }; // await criarContaReal(dados); // };
